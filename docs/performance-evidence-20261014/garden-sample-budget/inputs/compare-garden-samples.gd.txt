func _inverse_guard() -> bool:
	var spec: Variant=JSON.parse_string(FileAccess.get_file_as_string(PREFIX+"source-transform.json"))
	if not _check(spec is Dictionary and spec.get("schema")=="hex.garden-sample-transform.v1","Exact source transform manifest"): return false
	for row: Dictionary in spec.files:
		var before := FileAccess.get_file_as_string(row.before).replace("\r\n","\n")
		if not _check(before==FileAccess.get_file_as_string(row.canonical).replace("\r\n","\n"),"Canonical predecessor unchanged"): return false
		var transformed := FileAccess.get_file_as_string(row.candidate).replace("\r\n","\n")
		var changes: Array=row.changes.duplicate();changes.reverse()
		for edit: Dictionary in changes:
			if not _check(transformed.count(edit.new)==1,"Unique exact inverse block"): return false
			transformed=transformed.replace(edit.new,edit.old)
		if not _check(transformed==before,"Only declared candidate transformations"): return false
	return true

			okay=board.presenter.prepare_arrival(entries,required,reduced) and okay

				and int(board.presenter.arrival_status().prepared)==index+1,"One funded/visual staged row per frame, no early publication"): return false

		var started := Time.get_ticks_usec()

	groups.append({"index":index,"steps":count,"actual_order":order,"arms":arms})

	return _check(_markers(plain.model)==_markers(candidate.model),"Every group current lifecycle/phase/pulse state agrees")

func _minimum_cycle() -> int:

	return value if not plain.model._resident.is_empty() else -1

		"authority_hash_checked":proof or warning_changed,"memory":_memory() if save else {}})

func _sample_parity(limit: int) -> bool:
	var a: Dictionary=plain.presenter._surface._cache;var b: Dictionary=candidate.presenter._surface._cache
	if not _check(a.keys()==b.keys(),"Exact current surface mask ownership"): return false
	for mask: int in a:

		if not _check(int(right.sample_limit)==limit and right.samples==left.samples.slice(0,mini(limit,left.samples.size())),"Exact candidate path prepares only first successful public samples"): return false
		if not _check(left.style==right.style and left.sample_cells==right.sample_cells and left.boundary==right.boundary
			and left.cells==right.cells and left.native_opacity==right.native_opacity and left.native_entry==right.native_entry,"Full footprint/boundary/first-sample warning/style/native contract equal"): return false
	return true

func _markers(model: RefCounted) -> Dictionary:
	var result := {}
	for mask: int in model._resident:
		var entry: Dictionary=model._resident[mask]
		var court: HexChemistryWorld=entry.court
		var pulses := {}
		for id: int in court.products: pulses[str(id)]=int(court.products[id].get("pulse_index",-1))
		result[mask]={"phase":String(model._phase(entry)),"material":String(entry.material),"tick":int(entry.tick),"active":int(entry.active),"expiry":int(entry.expiry),"cycle":int(entry.cycle),
			"products":court.products.size(),"revision":court.value_revision,"render_revision":court.render_revision,"pulses":pulses}
	return result

func _transitions(before: Dictionary,after: Dictionary) -> Dictionary:
	var result := {"active_crossings":0,"expiry_crossings":0,"restarts":0,"product_pulse_mutations":0,"value_revision_changes":0,"render_revision_changes":0}
	for mask: int in before:
		if not after.has(mask): continue
		var a: Dictionary=before[mask];var b: Dictionary=after[mask]
		if int(a.tick)<int(a.active) and int(b.tick)>=int(a.active): result.active_crossings+=1
		if int(a.tick)<int(a.expiry) and int(b.tick)>=int(a.expiry): result.expiry_crossings+=1
		result.restarts+=int(b.cycle)-int(a.cycle)
		if int(a.revision)!=int(b.revision): result.value_revision_changes+=1
		if int(a.render_revision)!=int(b.render_revision): result.render_revision_changes+=1
		for id: String in a.pulses:
			if b.pulses.has(id) and int(a.pulses[id])!=int(b.pulses[id]): result.product_pulse_mutations+=1
	return result

func _state(model: RefCounted) -> Dictionary:
	var rows: Array[Dictionary]=[]
	for mask: int in model._resident:
		var entry: Dictionary=model._resident[mask]
		var court: HexChemistryWorld=entry.court
		var row := {"mask":mask,"court_hash":court.state_hash(),"conservation":court.conservation_valid(),"court_script":court.get_script().resource_path}
		for key: String in ["tick","cycle","started","active","decay","expiry","mode","material"]: row[key]=entry[key]
		rows.append(row)
	return {"rows":rows,"queue":model._restart_queue.duplicate(),"adoption":model._arrival_first_step.keys(),"budget":model._frame_restart_budget,"error":model.last_error}

func _equivalent(label: String) -> bool:

	return true

func _pixels_equal(expect_ink: bool) -> bool:
	var a := plain_view.get_texture().get_image();var b := candidate_view.get_texture().get_image()
	if not _check(a!=null and b!=null,"Native image readback"): return false
	a.convert(Image.FORMAT_RGBA8);b.convert(Image.FORMAT_RGBA8)
	if not _check(a.get_size()==SIZE and b.get_size()==SIZE and a.get_data()==b.get_data(),"Exact independent plain/candidate RGBA"): return false
	var ink := 0
	for y: int in range(0,SIZE.y,8):
		for x: int in range(0,SIZE.x,8):
			if a.get_pixel(x,y).to_rgba32()!=Color("283e37").to_rgba32(): ink+=1
	return _check(ink>0 if expect_ink else ink==0,"Expected actual Garden ink or complete departure")

func _weak_owners() -> Array[WeakRef]:
	var result: Array[WeakRef]=[]
	for board: Board in [plain,candidate]:
		for entry: Dictionary in board.model._resident.values(): result.append(weakref(entry.court))
		for row: Dictionary in board.presenter._surface._cache.values():
			for key: String in ["mesh","native_mesh"]:
				if row.get(key)!=null: result.append(weakref(row[key]))
		for retained: Dictionary in board.presenter._surface._last_geometry.values():
			var geometry: Dictionary=retained.geometry
			for key: String in ["mesh","native_mesh"]:
				if geometry.get(key)!=null: result.append(weakref(geometry[key]))
	return result

func _method(source: String,name: String) -> String:
	var start := source.find("func "+name+"(")
	if start<0: return ""
	var end := source.find("\nfunc ",start+1)
	return source.substr(start,end-start if end>=0 else -1).strip_edges()

func _coverage(world: SimWorld) -> Dictionary:
	var phases: Dictionary={};var owners: Dictionary={};var materials: Dictionary={};var regions:=0
	for cell: Vector2i in world.hex_court.occupied_cells():
		var value: Dictionary=world.hex_court.cell_record_value(cell)
		for key: Variant in value:
			if typeof(key) not in [TYPE_STRING,TYPE_STRING_NAME] or not value[key] is Dictionary or not String(key) in Court.MATERIALS: continue
			var material := String(key)
			var rows: Array[Dictionary]=world.hex_court.render_regions(cell,material,float(world.tick-1))
			for row: Dictionary in rows:
				regions+=1;phases[material+"/"+String(row.phase)]=true;owners[str(int(row.owner))]=true
				materials[material]=int(materials.get(material,0))+1
	var products: Array[Dictionary]=[]
	for product: Dictionary in world.hex_court.products.values():
		products.append({"id":int(product.id),"recipe_wire_id":int(product.recipe_wire_id),"material":String(product.material),"owner":int(product.owner),
			"source_ids":product.source_ids.duplicate(),"created":int(product.created),"active_tick":int(product.active_tick),"decay_tick":int(product.decay_tick),"expiry_tick":int(product.expiry_tick)})
	return {"regions":regions,"occupied_cells":world.hex_court.occupied_cells().size(),"phases":phases.keys(),"owners":owners.keys(),"materials":materials,"products":products}

func _fire_route(dense: bool) -> bool:
	var world := _world(FIRE)
	if world==null: return false
	var record := _record("twelve_fire" if dense else "single_fire_then_natural_reentry")
	var first_expired := false
	var max_active := -1
	for _step_index: int in 1500:
		var limit := 12 if dense else 2 if first_expired else 1
		var pressed := 0
		if int(record.attempts)<limit and _ready(world.player(),FIRE):
			pressed=SimCommand.PRESSED_SPELL_1;record.attempts+=1;record.attempt_ticks.append(world.tick)
		var commands: Array[SimCommand]=[SimCommand.new(world.tick,1,0,0,0,pressed,1000,0,1736000,880000)]
		if not _step(world,commands,record): return false
		var coverage := _coverage(world)
		if dense:
			for phase: String in ["forming","decaying"]:
				var label := "conflagration_"+phase
				if not snapshots.has(label) and ("conflagration/"+phase) in coverage.phases:
					if not _capture(label,world,coverage): return false
			if "conflagration/active" in coverage.phases and int(coverage.regions)>max_active:
				max_active=int(coverage.regions)
				if not _capture("fire_dense",world,coverage): return false
		else:
			for phase: String in ["ignition","active","cooling"]:
				var label := "fire_"+phase
				if not snapshots.has(label) and ("fire/"+phase) in coverage.phases:
					if not _capture(label,world,coverage): return false
			if first_expired and not snapshots.has("paid_reentry") and int(coverage.regions)>0:
				if not _capture("paid_reentry",world,coverage): return false
		if int(record.attempts)==limit and _empty(world):
			if not dense and not first_expired:
				first_expired=true
				if not _capture("expired",world,coverage): return false
			else: record.fully_expired=true;break
	var expected := 12 if dense else 2
	return _check(int(record.attempts)==expected and int(record.admitted)==expected and int(record.refused)==0
		and int(record.matter_paid)==expected*600 and bool(record.fully_expired),"Fire paid admissions/conservation/natural expiry")

func _mantle_route() -> bool:
	var wire := MixtureCatalog.wire_for(63,"mantle")
	var world := _world(wire)
	if world==null: return false
	var cast: Dictionary=CombatTuning.cast_definition_view(wire)
	var recipe: Dictionary=HexRecipePolicy.runtime_definition(HexRecipePolicy.wire_for_mask(63))
	var spacing := world.config.milliseconds_to_ticks(int(cast.startup_ms))+maxi(world.config.milliseconds_to_ticks(int(cast.cooldown_ms)),world.config.milliseconds_to_ticks(int(cast.recovery_ms)))+2
	var tail := maxi(int(recipe.formation_ticks)+int(recipe.active_ticks)+int(recipe.decay_ticks),HexChemistryWorld.trail_lifetime(int(cast.element_wire_id)))
	var horizon := 12*spacing+world.config.milliseconds_to_ticks(int(cast.lifetime_ms))+tail+2
	if not _check(horizon>0 and horizon<=4000,"Derived bounded Mantle route"): return false
	var record := _record("twelve_all_six_mantle_attempts")
	record.horizon=horizon;record.spacing_bound=spacing;record.tail_bound=tail
	var max_active := -1
	for _step_index: int in horizon:
		var pressed := 0
		if int(record.attempts)<12 and _ready(world.player(),wire):
			pressed=SimCommand.PRESSED_SPELL_1;record.attempts+=1;record.attempt_ticks.append(world.tick)
		var commands: Array[SimCommand]=[SimCommand.new(world.tick,1,0,0,0,pressed,1000,0,1736000,880000)]
		if not _step(world,commands,record): return false
		var coverage := _coverage(world)
		var active := false;var decaying := false
		var render_tick := world.tick-1
		for product: Dictionary in coverage.products:
			if int(product.recipe_wire_id)!=HexRecipePolicy.wire_for_mask(63): continue
			active=active or (render_tick>=int(product.active_tick) and render_tick<int(product.decay_tick) and (String(product.material)+"/active") in coverage.phases)
			decaying=decaying or (render_tick>=int(product.decay_tick) and render_tick<int(product.expiry_tick) and (String(product.material)+"/decaying") in coverage.phases)
		if active and int(coverage.regions)>max_active:
			max_active=int(coverage.regions)
			if not _capture("mantle_active",world,coverage,Vector2(1536,880)): return false
		if decaying and not snapshots.has("mantle_decaying"):
			if not _capture("mantle_decaying",world,coverage,Vector2(1536,880)): return false
		if int(record.attempts)==12 and _empty(world): record.fully_expired=true;break
	return _check(int(record.attempts)==12 and int(record.admitted)>0 and int(record.admitted)+int(record.refused)==12
		and int(record.matter_paid)>0 and bool(record.fully_expired),"Mantle attempts retain genuine payment/refusals and finite expiry")

		if not _check(int(candidate.presenter.get("_prepared_occupied_court_id"))==0,"One-call membership scope is consumed/restored after every draw"): return false
		if not _check(bool(candidate.last.path_eligible) and not bool(plain.last.path_eligible)
			and (not timed or int(candidate.last.prepared_cells)>0),"Pristine exact candidate/court/filter path; timed material reaches nonempty bundle"): return false
		var after_stats := {"plain":plain.presenter.outline_cache_stats(),"candidate":candidate.presenter.outline_cache_stats()}
		if not _check(after_stats.plain==after_stats.candidate,"Candidate preserves every retained boundary cache counter"): return false
		var row := {"index":index,"actual_order":trace.duplicate(),"plain":plain.last.duplicate(true),"candidate":candidate.last.duplicate(true),
			"cache_before":before_stats,"cache_after":after_stats,"new_topologies":int(after_stats.plain.topology_builds)-int(before_stats.plain.topology_builds),"new_meshes":int(after_stats.plain.uploads)-int(before_stats.plain.uploads)}
		if timed and index<WARMUPS: warmups.append(row)
		else: rows.append(row)
		if index==0 and not _pixels_equal(label!="expired"): return false
	if not _check(var_to_bytes(Packet.capture(world))==packet and var_to_bytes(world.combat_events)==events,"No drawing writes to packet/events"): return false
	if not _pixels_equal(label!="expired"): return false
	var images := {};var id := label+"_"+("reduced" if reduced else "normal")
	for role: String in ["plain","candidate"]:
		var image: Image=(plain_view if role=="plain" else candidate_view).get_texture().get_image();image.convert(Image.FORMAT_RGBA8)
		var path := output.path_join(id+"_"+role+".png")
		if not _check(image.save_png(path)==OK,"Save bounded representative PNG"): return false
		images[role]={"path":path,"sha256":FileAccess.get_sha256(path)}
	results.append({"label":id,"timed":timed,"reduced":reduced,"zoom":zoom,"jitter":[jitter.x,jitter.y],"focus":[plain.focus.x,plain.focus.y],
		"physical_size":[SIZE.x,SIZE.y],"world_tick":world.tick,"render_tick":world.tick-1,"coverage":snapshot.coverage,"matter_paid":snapshot.matter_paid,
		"authority_hash":str(state),"exact_rgba_first_and_final":true,"warmups":warmups,"rows":rows,"images":images})
	return true

func _pixels_equal(expect_ink: bool) -> bool:
	var a := plain_view.get_texture().get_image();var b := candidate_view.get_texture().get_image()
	if not _check(a!=null and b!=null,"Actual native images"): return false
	a.convert(Image.FORMAT_RGBA8);b.convert(Image.FORMAT_RGBA8)
	if not _check(a.get_size()==SIZE and b.get_size()==SIZE and a.get_data()==b.get_data(),"Exact plain/candidate RGBA"): return false
	if not expect_ink: return true
	var ink := 0
	for y: int in range(0,SIZE.y,4):
		for x: int in range(0,SIZE.x,4):
			var color := a.get_pixel(x,y).to_rgba32()
			if color!=Color("283e37").to_rgba32() and color!=Color("44504b").to_rgba32(): ink+=1
	return _check(ink>4,"Nonblank actual paid ground survives native crop")

func _check(condition: bool,message: String) -> bool:
	checks+=1
	if not condition: errors.append(message)
	return condition

func _memory_endpoint() -> Dictionary:
	return {"static":int(Performance.get_monitor(Performance.MEMORY_STATIC)),
		"textures":int(Performance.get_monitor(Performance.RENDER_TEXTURE_MEM_USED)),
		"video":int(Performance.get_monitor(Performance.RENDER_VIDEO_MEM_USED))}

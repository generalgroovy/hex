"""Read two completed native receipts; write one fresh bounded analysis artifact."""
from pathlib import Path
import argparse
import hashlib
import json
import math
import statistics

BASE = Path(__file__).resolve().parent
PRIMARY = {"fire_dense_normal", "fire_dense_reduced", "mantle_active_normal", "mantle_active_reduced"}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def stats(values):
    values = sorted(values)
    assert values and all(isinstance(v, (int, float)) and not isinstance(v, bool) and math.isfinite(v) for v in values)
    return {"n": len(values), "mean": statistics.mean(values), "median": statistics.median(values),
            "p95_nearest_rank": values[math.ceil(.95 * len(values)) - 1], "max": values[-1], "min": values[0]}


def group(rows):
    p = stats([r["plain"]["ground_us"] for r in rows])
    c = stats([r["candidate"]["ground_us"] for r in rows])
    delta = stats([r["plain"]["ground_us"] - r["candidate"]["ground_us"] for r in rows])
    return {"plain_us": p, "candidate_us": c, "paired_saving_us": delta,
            "mean_saving_fraction": (p["mean"]-c["mean"])/p["mean"] if p["mean"] else None,
            "candidate_p95_change_fraction": (c["p95_nearest_rank"]-p["p95_nearest_rank"])/p["p95_nearest_rank"] if p["p95_nearest_rank"] else None}


def load(path, reverse):
    record = json.loads(path.read_text(encoding="utf-8"))
    assert record["schema"] == "hex.ground.occupied-render-membership.v1" and record["passed"] is True and record["errors"] == []
    assert record["reverse_order"] is reverse and len(record["cases"]) == 22
    assert record["cleanup"] == {"viewports_created": 2, "viewports_freed": 2, "live_viewports": 0}
    result = {}
    for case in record["cases"]:
        if not case["timed"]:
            assert len(case["rows"]) == 1 and case["warmups"] == []
            continue
        rows = case["rows"]
        assert len(rows) == 32 and len(case["warmups"]) == 4
        by_order = {}
        for role in ("plain", "candidate"):
            selected = [r for r in rows if r["actual_order"][0] == role]
            assert len(selected) == 16
            by_order[role+"_first"] = group(selected)
        for r in rows:
            assert r["candidate"]["path_eligible"] is True and r["plain"]["path_eligible"] is False
            assert r["candidate"]["prepared_cells"] > 0
        row = group(rows)
        row["order"] = by_order
        row["warmups"] = group(case["warmups"])
        row["measured_new_topologies"] = sum(r["new_topologies"] for r in rows)
        row["measured_new_meshes"] = sum(r["new_meshes"] for r in rows)
        row["largest_candidate_regressions"] = [{"index": r["index"], "actual_order": r["actual_order"], "plain_us": r["plain"]["ground_us"], "candidate_us": r["candidate"]["ground_us"]} for r in sorted(rows, key=lambda r: r["candidate"]["ground_us"]-r["plain"]["ground_us"], reverse=True)[:3]]
        row["coverage"] = case["coverage"]
        row["matter_paid"] = case["matter_paid"]
        if case["label"] in PRIMARY:
            row["primary_gate"] = (row["paired_saving_us"]["mean"] >= 50 and row["mean_saving_fraction"] >= .03
                and row["paired_saving_us"]["median"] > 0
                and all(x["paired_saving_us"]["mean"] > 0 for x in by_order.values()))
        result[case["label"]] = row
    assert len(result) == 12 and PRIMARY <= result.keys()
    return record, result


parser = argparse.ArgumentParser()
parser.add_argument("--forward", required=True, type=Path)
parser.add_argument("--reverse", required=True, type=Path)
parser.add_argument("--output", required=True, type=Path)
args = parser.parse_args()
for path in (args.forward, args.reverse, args.output):
    assert path.resolve().is_relative_to(BASE), "Analysis is confined to the experiment."
assert not args.output.exists()
forward, a = load(args.forward, False)
reverse, b = load(args.reverse, True)
assert forward["source_hashes"] == reverse["source_hashes"] and forward["source_pins_sha256"] == reverse["source_pins_sha256"]
assert forward["qualification"] == reverse["qualification"]
gate = json.loads((BASE/"acceptance-before-timing.json").read_text())
assert gate["source_pins_sha256"] == forward["source_pins_sha256"]
primary_pass = all(process[name]["primary_gate"] for process in (a, b) for name in PRIMARY)
tail_regressions = [name for name in a.keys()-PRIMARY if a[name]["candidate_p95_change_fraction"] > .05 and b[name]["candidate_p95_change_fraction"] > .05]
output = {"schema": "hex.membership-paired-analysis.v1", "receipt_sha256": {"forward": sha(args.forward), "reverse": sha(args.reverse)},
          "analyzer_sha256": sha(Path(__file__)), "predeclared_gate_sha256": sha(BASE/"acceptance-before-timing.json"),
          "source_pins_sha256": forward["source_pins_sha256"], "qualification": forward["qualification"],
          "forward": a, "reverse": b, "all_primary_case_process_order_gates_pass": primary_pass,
          "repeated_other_case_p95_regressions_over_5_percent": tail_regressions,
          "meets_declared_microbenchmark_gate": primary_pass and not tail_regressions,
          "decision_scope": "Passing this screen permits review only, not automatic canonical promotion. Paired phase-correlated native draw elapsed samples are not GPU/FPS or current full-profiler speed evidence. Warmups and invocation-order strata remain separate; p95 uses nearest rank; positive savings mean lower candidate elapsed."}
encoded = json.dumps(output, indent=2) + "\n"
with args.output.open("x", encoding="utf-8", newline="\n") as handle:
    handle.write(encoded)
assert args.output.read_text() == encoded
print(json.dumps({"output": str(args.output), "sha256": sha(args.output), "meets_declared_microbenchmark_gate": output["meets_declared_microbenchmark_gate"]}))
